@startuml DesignCLD_Train
!pragma ratio 0.6
skinparam groupInheritance 2

namespace SimpleTrainSimCld.Train {
    class Train {
        - weight : int
        - axles_count : int
        - /speed : double
        - throttle_pos : double
        - brake_lever_pos : int
        - /braking_force : double
        - aerodynamic_efficiency : double

        + get_weight() : int
        + get_axles_count() : int
        + get_speed() : double
        + get_acceleration() : double
        + get_engine_power() : double
        + get_braking_force() : double
        + change_power(throttle_lever_pos) : void
        + change_brake(brake_lever_pos) : void
    }

    abstract class Engine {
        - max_power : double
        - /current_power : double

        + calculate_power(throttle_lever_pos) : double
    }

    class "Electric engine" as electric_engine {}

    class "Combustion engine" as combustion_engine {
        - fuel_level : double
        - max_force_fuel_burning : double

        + get_fuel() : double
        + add_fuel(ammount) : void
        + burn_fuel() : void
    }

    class "Electric drive CE" as ce_electric_drive {}

    class "Direct drive CE" as ce_direct_drive {
        - max_torque : double
        - /current_torque : double
        + calculate_torque() : double
    }

    class Brake {
        - max_force : double
        - current_force : double

        + calculate_force(brake_lever_pos) : double
    }

    class "Deadman's switch" as deadmans_switch {
        - interval : int
        - state : DeadmansSwState

        - count() : void
        - activate_brakes() : void
        + acknowledge() : void
    }

    enum DeadmansSwState {
        DISABLED
        VIGILANT_QUIET
        VIGILANT_WARNING
        TRIGGERED
    }
    DeadmansSwState .. deadmans_switch

    Train "1" *-up- "1" deadmans_switch : secures <
    Train "1" *-up- "1" Engine : powers <
    Train "1" *-up- "1" Brake : slows <
    Engine <|-up- electric_engine
    Engine <|-up- combustion_engine
    combustion_engine <|-- ce_electric_drive
    combustion_engine <|-- ce_direct_drive : {complete, disjoint}
}
@enduml

@startuml DesignCLD_UI

namespace SimpleTrainSimCld.UI {
    class HUD {
        - layout : int

        + draw() : void
    }

    abstract class Indicator {
        - width : int
        - height : int
        - shape : int
        - resolution : int

        + get_value() : void
    }

    class "Speed indicator" as speed_indicator {
        - current_speed : double
    }

    class "Throttle indicator" as throttle_indicator {
        - type : IndicatorType
    }

    class "Braking indicator" as braking_indicator {
        - type : IndicatorType
    }

    enum IndicatorType {
        LEVER_SETTING
        FORCE
    }
    IndicatorType .. throttle_indicator
    IndicatorType .. braking_indicator

    class "Wheelslip indicator" as wheelslip_indicator {}

    class "Gradient indicator" as gradient_indicator {}

    class "Acceleration indicator" as acceleration_indicator{}

    HUD "1" *-- "*" Indicator : consists of >

    Indicator <|-- speed_indicator
    Indicator <|-- throttle_indicator
    Indicator <|-- braking_indicator
    Indicator <|-- wheelslip_indicator
    Indicator <|-- gradient_indicator
    Indicator <|-- acceleration_indicator : {complete,disjoint}
}
@enduml

namespace SimpleTrainSimCld.Route {
    class Route {
        - {static} rotation = 0deg : int
        - /length : int

        + get_slope(start, end) : double
    }

    class "Route point" as route_point {
        - position : int
        - altitude : int

        + get_position() : int
        + get_altitude() : int
    }

    class Environment {
        - /adhesion_modifier : double

        + get_adhesion_modifier() : double
    }

    class Dampness {
        - track_dampness : int {0-100}

        + get_dampness() : int
        + change_dampness(value) : void
    }

    class Temperature {
        - value : double

        + get_value() : double
        + change_value(value) : void
    }

    class Wind {
        - strength : int
        - direction : int

        + get_strength() : int
        + get_direction() : int
        + change_strength(speed) : void
        + change_direction(deg) : void
    }

    class Leaves {
        - coverage : int {0-100}

        + get_coverage() : int
        + change_coverage(value) : int
    }

    Route "1" *-down- "*" route_point : consists of >
    Route "1" -right- "*" Environment : impacts <
    route_point "1" -right- "*" Environment  : starting point <
    route_point "1" -right- "*" Environment : ending point <

    Environment <|-- Dampness : {complete,disjoint}
    Environment <|-- Temperature
    Environment <|-- Wind
    Environment <|-- Leaves
}

namespace SimpleTrainSimCld.Physics {
    class "Physical force" as physical_force

        - {static} acceleration = 9.81m/s<sup>2</sup> : double
    
        + calculate_effect(slope) : double
    }

    }

    physical_force <|-- Gravity : {complete,disjoint}
    physical_force <|-- aerodynamic_drag

    physical_force <|-- mechanical_resistance
    physical_force <|-- engine_force
}
